\section{Problemas}

\begin{enumerate}
	\item Usando la ecuación Shockley:
	
		\[
		I = I_S (e^{\frac{V_D}{V_T}}-1)
		\]
		\[
		V_D = V_T * ln(\frac{I}{I_S}+1)
		\]
		
		Como el primer diodo está en polarización inversa, se toma $I=I_S$:
		
		\[
		V_1 = V_T * ln(2) = \frac{18+273.15}{11600} * ln 2 = 0.017V
		\]
		
		Aplicando la segunda ley de Kirchoff:
		
		\[
		5V-V_1-V_2 = 0
		\]
		\[
		V_2 = V_1-5V = 0.017V-5V = -4.98
		\]
		
	\item Usando la segunda aproximación del diodo se tiene que el diodo 1 se reemplaza por una caida de potencial por estar en polarización directa y el diodo 2 por un interruptor abierto:
	
		\begin{center}
			\begin{circuitikz}
				\draw
				(0,0) to [R, *-*, l={$R=2.2k\Omega$}, i_=$I$] (4,0)
				to [short] (4,1.5)
				to [battery1, l_=$D_1$, v={$-0.7V$}, invert] (6.5,1.5)
				to [short] (6.5,-1.5)
				to [short] (5.75,-1.5)
				to [short, *-*, l=$D_2$] (5,-2)
				
				(5,-1.5) to [short] (4,-1.5)
				to [short] (4,0)
				
				(6.5,0) to [short, *-*] (8,0)
				
				(0,0) node[anchor=north]{$E_1=20V$}
				(4,0) node[anchor=west]{$a$}
				(6.5,0) node[anchor=east]{$b$}
				(8,0) node[anchor=north]{$E_2=-4V$};
			\end{circuitikz}
		\end{center}
		
		El voltaje en el punto inicial es de $E_1 = 20V$ y en el punto final es $E_2 = -4V$, por lo que hubo una caida de voltaje de $24V$ entre estos puntos, de la cual $0.7V$ son debidos al diodo, por lo que entre el punto inicial y el nodo $a$ hay una caida de $23.3V$, que es el voltaje en la resistencia.
		
		Usando ley de Ohm:
		
		\[
		I = \frac{V}{R} = \frac{23.3V}{2200 \Omega} = 10.59 mA
		\]
		
	\item 
		\begin{description}
		\item[a.] 
		
		Para el primer circuito, hay un voltaje de 0,7V en el diodo de silicio por lo que $V_{o1} = 11.3V$, además, la corriente del circuito está dada por:
		
		\[
		I = \frac{12V - 0.7V - 0.3V}{4.7K\Omega} = 2.34 mA
		\]
		
		En la resistencia:
		
		\[
		V_R = 2.34mA * 4.7k\Omega = 11V
		\]
		
		Por lo que $V_{o2} = V_{o1} - V_R = 0.3V$
		
		\item[b.]
		
		Para el segundo circuito, la corriente se calcula con:
		
		\[
		I = \frac{10V - 0.7V - 0,3V}{1.2k\Omega + 3.3k\Omega} = 2mA
		\]
		
		Para la resistencia de $3.3k\Omega$:
		
		\[
		V = 3.3k\Omega * 2mA = 6.6V
		\]
		
		Por lo tanto, $V_{o2} = -6.6V$
		
		Para la resistencia de $1.2k\Omega$:
		
		\[
		V = 1.2k\Omega * 2mA = 2.4V
		\]
		
		Por lo tanto, $V_{o2} = -2.4V$
		
		\end{description}
		
	\item \begin{description}
		\item[a.] 
		
		La corriente en el circuito está dada por:
		
		\[
		I = \frac{20V - 0.7V}{4.7k\Omega} = 4.1mA
		\]
		
		La corriente se divide de igual manera para los dos diodos en paralelo, ya que ambos son de silicio y tienen polarización directa, entonces $I_D = I/2 = 2.05mA$
		
		$V_o$ se calcula después de la caida de potencial de los diodos, por lo tanto $V_o = 19.3V$.
		
		\item[b.]
		
		La corriente en el circuito está dada por:
		
		\[
		I = \frac{20V - 0.7V}{2.2k\Omega} = 8.77mA
		\]
		
		La corriente solo pasa por un diodo porque es el único en polarización directa.
		
		$V_o$ se calcula después de la caida de potencial del diodo, entonces $V_o = 14.3V$.
		
	\end{description}
\end{enumerate}
