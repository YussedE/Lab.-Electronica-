	\item \begin{enumerate}
		\item 
		
		\begin{proof}
			
			Para una señal de corriente alterna senoidal pura, la ecuación temporal del voltaje es:
			
			\begin{equation*}
				v(t) = V_{p}\sin(\omega t)
			\end{equation*}
			
			Donde $V_{p}$ es el voltaje pico, $T$ es el periodo y $\omega = \frac{2\pi}{T}$ es la frecuencia angular. Sustituyendo esto en la definición de $V_{rms}$:
			
			\begin{equation*}
				V_{rms}^{2} = \frac{1}{T}\int_{0}^{T}[V_{p}\sin(\omega t)]^{2} dt = \frac{V_{p}^{2}}{T}\int_{0}^{T} \sin^{2}(\omega t) dt
			\end{equation*}
			
			Aplicando la identidad trigonométrica del ángulo doble:
			
			\begin{equation*}
				\sin^{2}(\theta) = \frac{1 -  \cos(2\theta)}{2}
			\end{equation*}
			
			Sustituyendo $\theta = \omega t$:
			
			\begin{equation*}
				V_{rms}^{2} = \frac{V_{p}^{2}}{T} \int_{0}^{T} \frac{1 - \cos(2\omega t)}{2} dt = \frac{V_{p}^{2}}{2T} \left[\int_{0}^{T}dt - \int_{0}^{T}\cos(2\omega t)dt\right]
			\end{equation*}
			
			Evaluando las integrales:
			
			\begin{equation*}
				\int_{0}^{T} dt = T
			\end{equation*}
			
			\begin{equation*}
				\int_{0}^{T} \cos(2\omega t) dt = 0
			\end{equation*}
			
			Sustituyendo los valores de las integrales en la ecuación:
			\begin{equation*}
				V_{rms}^{2} = \frac{V_{p}^{2}}{2T}(T) = \frac{V_{p}^{2}}{2}
			\end{equation*}
			
			Despejando $V_{rms}$ al aplicar la raíz cuadrada en ambos lados de la ecuación:
			\begin{equation*}
				V_{rms} = \frac{V_{p}}{\sqrt{2}}
			\end{equation*}
			
		\end{proof}
		
		\item \begin{equation}
			\tan(\theta) = \frac{\text{Cateto Opuesto}}{\text{Cateto Adyacente}} = \frac{-X_C}{R}
			\label{angulo-desfase}
		\end{equation}
		
		Despejando el ángulo mediante la función inversa de la arcotangente, se obtiene la ecuación fundamental del desfase:
		\begin{equation}
			\theta = \arctan\left(\frac{-X_C}{R}\right) = -\arctan\left(\frac{1}{\omega C R}\right)
		\end{equation}
		
		El signo negativo resultante posee un significado físico crítico: confirma matemáticamente que en un circuito puramente capacitivo o RC, \textbf{el voltaje se encuentra retrasado} respecto a la corriente (o equivalentemente, la corriente adelanta al voltaje) en un rango angular comprendido entre $-90^\circ < \theta < 0^\circ$.

	\item El voltaje rms de una corriente alterna es el valor equivalente que proporciona la misma cantidad de energía en una corriente directa; el voltaje pico es el punto máximo al que llega al voltaje o la amplitud de la función; el voltaje pico a pico abarca desde el punto menor hasta el punto mayor de la función.
	
	En una onda senoidal se relacionan así:
	\[
	V_{rms} = \frac{V_{pp}}{2\sqrt{2}} = \frac{V_{p}}{\sqrt{2}}
	\]
	
	En una onda senoidal se relacionan así:
	\[
	V_{rms} = \frac{V_{pp}}{2} = V_{p}
	\]
