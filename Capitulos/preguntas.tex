\section{Preguntas complementarias}

\begin{enumerate}
	
    \item Primer circuito
    
    \begin{circuitikz}
    	\draw
    	(0,0) node[left] {A} to [short,*-] (0,0.7)
    	to [R, l^=$R_1$, a=$4\Omega$] (-3,0.7)
    	to [battery1, l_=$E_1$, a^=$28V$] (-3,-2.7)
    	to [short,-] (3,-2.7)
    	to [battery1, invert, l_=$E_2$, a^=$7V$] (3,0.7)
    	to [R, l^=$R_2$, a=$1\Omega$] (0,0.7)
    	(0,-2.7) to [short,-*] (0,-2) node[left] {B}
    	to [R, l^=$R_L$, a=$2\Omega$] (0,0);
    \end{circuitikz}
    
    \begin{description}
    	\item[a.] Para calcular la resistencia de Thévenin se considera el circuito de la siguiente manera:
    	
    	\begin{circuitikz}
    		\draw
    		(0,1.4) node[left] {A} to [short,*-] (0,0.7)
    		to [short,-] (-3,0.7)
    		to [R, l^=$R_1$, a=$4\Omega$] (-3,-2.7)
    		to [short,-] (3,-2.7)
    		to [R, l^=$R_2$, a=$1\Omega$] (3,0.7)
    		to [short,-] (0,0.7)
    		(0,-2.7) to [short,-*] (0,-3.4) node[left] {B};
    	\end{circuitikz}
    	
    	De esta manera, la resistencia $R_{TH}$ se expresa como:
    	
    	\[
    	R_{TH} = \frac{R_1 * R_2}{R_1 + R_2} = \frac{4}{5} \Omega
    	\]
    	
    	\item[b.] Para calcular el voltaje Thévenin se considera el circuito de la siguiente manera:
    	
    	\begin{circuitikz}
    		\draw
    		(0,1.4) node[left] {A} to [short,*-] (0,0.7)
    		to [R, l^=$R_1$, a=$4\Omega$] (-3,0.7)
    		to [battery1, l_=$E_1$, a^=$28V$, i=$I$] (-3,-2.7)
    		to [short,-] (3,-2.7)
    		to [battery1, invert, l_=$E_2$, a^=$7V$ ,i=$I$] (3,0.7)
    		to [R, l^=$R_2$, a=$1\Omega$] (0,0.7)
    		(0,-2.7) to [short,-*] (0,-3.4) node[left] {B};
    	\end{circuitikz}
    	
    	Por leyes de Kirchoff se obtiene:
    	
    	\[
    	E_1 - R_1*I - R_2*I - E_2 = 0
    	\]
    	\[
    	I = \frac{E_1-E_2}{R_1+R_2} = \frac{21}{5} A
    	\]
    	
    	La caida de potencial en cada una de las resistencias:
    	
    	\[
    	V_1 = I*R_1 = \frac{84}{5} V
    	\]
    	\[
    	V_2 = I*R_2 = \frac{21}{5} V
    	\]
    	
    	La caida de potencial entre A y B es igual a la caida de la resistencia $R_2$ y el voltaje de la fuente $E_2$:
    	
    	\[
    	V_{AB} = V_2 + E_2 = \frac{56}{5} V = V_{TH}
    	\]
    	
    	\item[c.] Para calcular la corriente que pasa por $R_L$:
    	
    	\[
    	I = \frac{V_{TH}}{R_{TH} + R_L} = 4 A
    	\]
    	
    	\item[d.] El circuito equivalente es:
    	
    	\begin{circuitikz}
    		\draw
    		(0,0) node[left] {A} to [short,*-] (0,0.7)
    		to [R, l^=$R_{TH}$, a=$\frac{4}{5}\Omega$] (-3,0.7)
    		to [battery1, l_=$E_{TH}$, a^=$\frac{21}{4}V$] (-3,-2.7)
    		to [short,-] (0,-2.7)
    		to [short,-*] (0,-2) node[left] {B}
    		to [R, l^=$R_L$, a=$2\Omega$] (0,0);
    	\end{circuitikz}
    	
    \end{description}
    
    \item Segundo circuto
    
    \begin{circuitikz}
    	\draw
    	(3,0) node[left] {A} to [short,*-] (3,0.7)
    	to [R, l^=$R_2$, a=$3K\Omega$] (0,0.7)
    	to [R, l^=$R_1$, a=$2K\Omega$] (-3,0.7)
    	to [battery1, l_=$E_1$, a^=$4V$] (-3,-2.7)
    	to [short,-] (3,-2.7)
    	to [short,-*] (3,-2) node[left] {B}
    	to [R, l^=$R_L$, a=$1K\Omega$] (3,0)
    	(0,-2.7) to [ammeter, l^=$I$, a=$2mA$] (0,0.7);
    \end{circuitikz}
    
    \begin{description}
    	
    	\item[b.] Para calcular el voltaje Thévenin se considera el circuito de la siguiente manera:
    	
    	\begin{circuitikz}
    		\draw
    		(3,0) node[left] {A} to [short,*-] (3,0.7)
    		to [R, l^=$R_2$, a=$3K\Omega$] (0,0.7)
    		to [R, l^=$R_1$, a=$2K\Omega$] (-3,0.7)
    		to [battery1, l_=$E_1$, a^=$4V$] (-3,-2.7)
    		to [short,-] (3,-2.7)
    		to [short,-*] (3,-2) node[left] {B}
    		(0,-2.7) to [ammeter, l^=$I$, a=$2mA$] (0,0.7);
    	\end{circuitikz}
    	
    	Por leyes de Kirchoff se obtiene:
    	
    	\[
    	E_1 - R_1*I + V_{TH} = 0
    	\]
    	\[
    	V_{TH} = R_1*I - E_1 = 16 V
    	\]
    		
    	\item[a.] Para calcular la resistencia de Thévenin se considera el circuito de la siguiente manera:
    	
    	\begin{circuitikz}
    		\draw
    		(3,0) node[left] {A} to [short,*-] (3,0.7)
    		to [R, l^=$R_2$, a=$3K\Omega$] (0,0.7)
    		to [R, l^=$R_1$, a=$2K\Omega$] (-3,0.7)
    		to [short,-] (-3,-2.7)
    		to [short,-] (3,-2.7)
    		to [short,-*] (3,-2) node[left] {B}
    		(0,-2.7) to [ammeter, l^=$I$, a=$2mA$] (0,0.7);
    	\end{circuitikz}
    	
    	
    	
    	La resistencia del amperímetro es:
    	
    	\[
    	R_A = \frac{V_A}{I_A} = \frac{V_{TH}}{I_A} = 8K \Omega
    	\]
    	
    	De esta manera, la resistencia $R_{TH}$ se expresa como:
    	
    	\[
    	R_{TH} = \frac{R_1 * R_A}{R_1 + R_A} + R_2= \frac{23}{5} K\Omega
    	\]
    	
    	\item[c.] Para calcular la corriente que pasa por $R_L$:
    	
    	\[
    	I = \frac{V_{TH}}{R_{TH}+R_L} = \frac{80}{23}mA
    	\]
    	
    	\item[d.] El circuito equivalente es:
    	
    	\begin{circuitikz}
    		\draw
    		(0,0) node[left] {A} to [short,*-] (0,0.7)
    		to [R, l^=$R_{TH}$, a=$\frac{23}{5}K\Omega$] (-3,0.7)
    		to [battery1, l_=$E_{TH}$, a^=$16V$] (-3,-2.7)
    		to [short,-] (0,-2.7)
    		to [short,-*] (0,-2) node[left] {B}
    		to [R, l^=$R_L$, a=$1K\Omega$] (0,0);
    	\end{circuitikz}
    	
    \end{description}

\end{enumerate}
