\section{Preguntas}

\begin{enumerate}
	\item Para calcular el $V_{DC}$ se usa el valor promedio:
	
	\[
	V_{DC} = \frac{1}{T} \int_{0}^{T} V(t)\,dt
	\]
	
	Para una señal de media onda $V_{DC} = (V_p-V_T) Sen(\omega t)$ para $0<t<T/2$ y $V_{DC} = 0$ para $T/2<t<T$, entonces:
	
	\[
	V_{DC} = \frac{1}{T} [\int_{0}^{T/2} (V_p-V_T) Sen(\omega t)\,dt + \int_{T/2}^{T} 0\,dt]
	\]
	
	\[
	V_{DC} = \frac{1}{T} (V_p-V_T) \frac{1}{\omega} [-Cos(\omega \frac{T}{2}) + Cos(0)]
	\]
	
	Pero $\omega = 2\pi / T$:
	
	\[
	V_{DC} = \frac{1}{T} (V_p-V_T) \frac{1}{2\pi / T} [-Cos(\frac{2\pi}{T} \frac{T}{2}) + Cos(0)]
	\]
	
	\[
	V_{DC} = \frac{V_p-V_T}{2\pi} [-Cos(\pi) + Cos(0)]
	\]
	
	\[
	V_{DC} = \frac{V_p-V_T}{\pi}
	\]
	
	\item Para un rectificador de 2 diodos $V_{DC} = (\frac{V_p}{2}-V_T) Sen(\omega t)$ para $0<t<T/2$, esta curva se repite en $T/2<t<T$, entonces:
	
	\[
	V_{DC} = \frac{2}{T} \int_{0}^{T/2} (\frac{V_p}{2}-V_T) Sen(\omega t)\,dt
	\]
	
	\[
	V_{DC} = \frac{2}{T} (\frac{V_p}{2}-V_T) \frac{1}{\omega} [-Cos(\omega \frac{T}{2}) + Cos(0)]
	\]
	
	\[
	V_{DC} = \frac{2}{T} (\frac{V_p}{2}-V_T) \frac{1}{2\pi / T} [-Cos(\frac{2\pi}{T} \frac{T}{2}) + Cos(0)]
	\]
	
	\[
	V_{DC} = \frac{\frac{V_p}{2}-V_T}{\pi} [-Cos(\pi) + Cos(0)]
	\]
	
	\[
	V_{DC} = 2\frac{V_p-V_T}{\pi}
	\]
	
	\item Para un rectificador de 4 diodos $V_{DC} = (V_p-2V_T) Sen(\omega t)$ para $0<t<T/2$, esta curva se repite en $T/2<t<T$, entonces:
	
	\[
	V_{DC} = \frac{2}{T} \int_{0}^{T/2} (V_p-2V_T) Sen(\omega t)\,dt
	\]
	
	\[
	V_{DC} = \frac{2}{T} (V_p-2V_T) \frac{1}{\omega} [-Cos(\omega \frac{T}{2}) + Cos(0)]
	\]
	
	\[
	V_{DC} = \frac{2}{T} (V_p-2V_T) \frac{1}{2\pi / T} [-Cos(\frac{2\pi}{T} \frac{T}{2}) + Cos(0)]
	\]
	
	\[
	V_{DC} = \frac{V_p-2V_T}{\pi} [-Cos(\pi) + Cos(0)]
	\]
	
	\[
	V_{DC} = 2\frac{V_p-2V_T}{\pi}
	\]
\end{enumerate}
